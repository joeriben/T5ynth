# T5ynth unified inference backend dependencies (ONE backend, git-main stack).
# A single interpreter serves both the diffusers models (Stable Audio Open 1.0,
# AudioLDM2) and the stable-audio-tools NATIVE models (SAO Small AND Stable
# Audio 3 Small Music). SA3 needs t5gemma (transformers >=4.53) plus the
# git-main stable-audio-tools (T5GemmaConditioner, diffusion_cond_inpaint,
# native sample_flow_pingpong, APG). transformers stays <5 so diffusers and
# audioldm2 keep working. Validated end-to-end (handshake + SA3 + SAO Small
# generation, determinism, peak normalisation) on exactly this set.
diffusers>=0.24.0
transformers>=4.53,<5
safetensors
accelerate
requests
# git main: the previously-pinned 0.0.19 cannot load SA3 at all (it has neither
# the t5gemma conditioner nor diffusion_cond_inpaint). git-main additionally
# requires the REAL pytorch_lightning — its DiT import path pulls
# models/lora/callbacks.py -> `import pytorch_lightning as pl` and uses
# pl.callbacks — so pipe_inference.py no longer mocks it and it is listed here.
stable-audio-tools @ git+https://github.com/Stability-AI/stable-audio-tools.git
pytorch-lightning

# numpy <2 so the PyWavelets==1.4.1 that stable-audio-tools hard-pins (numpy-1
# ABI) imports cleanly. numpy>=2 would need PyWavelets>=1.6, conflicting with
# sat's pin and forcing a post-install override; numpy 1.26 satisfies torch
# 2.7.1, scipy, transformers and diffusers, so the install resolves in one pass.
numpy>=1.23,<2
scipy
torchsde
einops
# protobuf <3.21 — needed ONLY on the path where a t5gemma tokenizer is loaded from
# a bare SentencePiece `tokenizer.model` with no `tokenizer.json` (the ungated SA3
# install): AutoTokenizer.from_pretrained (default use_fast=True) builds a
# GemmaTokenizerFast via the slow->fast SentencePiece converter, which imports
# sentencepiece 0.1.99's bundled *_pb2. Those generated descriptors are incompatible
# with protobuf >=4 ("Descriptors cannot be created directly"), so the converter only
# runs under protobuf 3.20.x. Harmless for the diffusers/SAO paths (they ship a
# tokenizer.json and never invoke the converter). Validated alongside a full SA3
# music generation (RMS 0.16, identical token IDs to the shipped tokenizer.json).
protobuf<3.21
# Keep native extension wheels ABI-aligned with the torch version installed by

torchaudio==2.7.1
